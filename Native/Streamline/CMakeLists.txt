cmake_minimum_required(VERSION 3.24)
project(ReduxBetterAA_StreamlineProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Streamline probe requires Windows x64")
endif()
set(STREAMLINE_SDK_ROOT "" CACHE PATH "External official Streamline v2.14.1 checkout")
if(NOT EXISTS "${STREAMLINE_SDK_ROOT}/include/sl_dlss_g.h")
    message(FATAL_ERROR "Set STREAMLINE_SDK_ROOT to the pinned external Streamline source checkout")
endif()
file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../.." source_root)
file(REAL_PATH "${STREAMLINE_SDK_ROOT}" sdk_root)
file(REAL_PATH "${CMAKE_BINARY_DIR}" build_root)
cmake_path(IS_PREFIX source_root "${sdk_root}" NORMALIZE sdk_inside_source)
cmake_path(IS_PREFIX source_root "${build_root}" NORMALIZE build_inside_source)
if(sdk_inside_source OR build_inside_source)
    message(FATAL_ERROR "SDK checkout and build output must remain outside the Better AA checkout")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${sdk_root}" rev-parse HEAD
    OUTPUT_VARIABLE sdk_commit OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT sdk_commit STREQUAL "2122257e0fce486f91b385aa63b9a09b0a34b363")
    message(FATAL_ERROR "Streamline headers must be pinned to v2.14.1 commit 2122257e0fce486f91b385aa63b9a09b0a34b363")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${sdk_root}" diff --exit-code HEAD -- include
    COMMAND_ERROR_IS_FATAL ANY OUTPUT_QUIET)
add_library(ReduxBetterAA.StreamlineProbe SHARED StreamlineProbe.cpp StreamlineProbe.h)
target_compile_definitions(ReduxBetterAA.StreamlineProbe PRIVATE RBA_STREAMLINE_EXPORTS)
target_include_directories(ReduxBetterAA.StreamlineProbe SYSTEM PRIVATE "${sdk_root}/include")
target_link_libraries(ReduxBetterAA.StreamlineProbe PRIVATE d3d11 d3d12 dxgi bcrypt wintrust crypt32)
add_library(ReduxBetterAA.StreamlineProvider SHARED StreamlineProvider.cpp StreamlineProvider.h
    StreamlineProviderState.h StreamlineProviderConstants.h StreamlineRuntime.h)
target_compile_definitions(ReduxBetterAA.StreamlineProvider PRIVATE RBA_SL_PROVIDER_EXPORTS)
target_include_directories(ReduxBetterAA.StreamlineProvider SYSTEM PRIVATE "${sdk_root}/include")
target_link_libraries(ReduxBetterAA.StreamlineProvider PRIVATE d3d11 d3d12 dxgi user32 bcrypt wintrust crypt32)
add_executable(StreamlineProbeTests StreamlineProbeTests.cpp)
target_link_libraries(StreamlineProbeTests PRIVATE ReduxBetterAA.StreamlineProbe d3d11)
add_executable(StreamlineProbeInternalTests StreamlineProbeInternalTests.cpp)
target_include_directories(StreamlineProbeInternalTests SYSTEM PRIVATE "${sdk_root}/include")
target_link_libraries(StreamlineProbeInternalTests PRIVATE d3d11 d3d12 dxgi bcrypt wintrust crypt32)
add_executable(StreamlineProviderTests StreamlineProviderTests.cpp)
target_include_directories(StreamlineProviderTests SYSTEM PRIVATE "${sdk_root}/include")
target_link_libraries(StreamlineProviderTests PRIVATE d3d11 d3d12 dxgi user32 bcrypt wintrust crypt32)
add_executable(StreamlineProviderGpuTests StreamlineProviderGpuTests.cpp)
target_link_libraries(StreamlineProviderGpuTests PRIVATE ReduxBetterAA.StreamlineProvider d3d11 d3d12 dxgi user32)
add_library(StreamlineModuleAuditFixture SHARED ModuleAuditFixture.cpp)
set_target_properties(StreamlineModuleAuditFixture PROPERTIES OUTPUT_NAME "sl.reflex"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/audit-fixture")
add_dependencies(StreamlineProbeInternalTests StreamlineModuleAuditFixture)
add_dependencies(StreamlineProviderTests StreamlineModuleAuditFixture)
foreach(target ReduxBetterAA.StreamlineProbe ReduxBetterAA.StreamlineProvider StreamlineProbeTests StreamlineProbeInternalTests StreamlineProviderTests StreamlineProviderGpuTests StreamlineModuleAuditFixture)
    target_compile_features(${target} PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endforeach()
enable_testing()
add_test(NAME StreamlineProbeContract COMMAND StreamlineProbeTests)
add_test(NAME StreamlineProbeLifecycleAndModuleAudit COMMAND StreamlineProbeInternalTests "$<TARGET_FILE:StreamlineModuleAuditFixture>")
add_test(NAME StreamlineProviderContracts COMMAND StreamlineProviderTests "$<TARGET_FILE:StreamlineModuleAuditFixture>")
