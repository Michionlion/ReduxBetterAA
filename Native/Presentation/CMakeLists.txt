cmake_minimum_required(VERSION 3.24)
project(ReduxBetterAAPresentation LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The presentation probe requires Windows x64 and MSVC")
endif()

set(UNITY_PLUGIN_API "" CACHE PATH "External Unity 6000.5.8f1 Editor/Data/PluginAPI directory")
# The D3D11 interface GUID is unversioned. Never compile against a shorter public
# sample header and then read the appended swapchain function pointers.
set(unity_headers IUnityInterface.h IUnityGraphics.h IUnityGraphicsD3D11.h IUnityRenderingExtensions.h)
set(unity_hashes
    e23e267b57c049b438307542d62c30fbcd3fa77b6dac3b08842eae0df10b8239
    524ca42ff0274feec5aea17a5275b774ffc1a031bdc9378f3e3eb2be787a7fa6
    873389c2bc367b7ef7be4809059133b692a1c03eb268cfebf5bcb290c511bace
    dbcb270c5065d9ce98492b4e35d5be91f0f2cfea37be08619c3e03a12604fc37)
foreach(index RANGE 0 3)
    list(GET unity_headers ${index} header)
    list(GET unity_hashes ${index} expected)
    if(NOT EXISTS "${UNITY_PLUGIN_API}/${header}")
        message(FATAL_ERROR "Pass UNITY_PLUGIN_API pointing at the exact Unity 6000.5.8f1 headers outside source")
    endif()
    file(SHA256 "${UNITY_PLUGIN_API}/${header}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Unity header ${header} differs from the inspected 6000.5.8f1 version")
    endif()
endforeach()

add_library(GfxPluginReduxBetterAAPresentation SHARED PresentationProbe.cpp PresentationProbe.h CompositionPresenter.cpp CompositionPresenter.h)
target_link_libraries(GfxPluginReduxBetterAAPresentation PRIVATE d3d11 d3d12 dxgi dcomp)
# Compile the same implementation into a test-only DLL for failure injection.
# The Unity-preloaded runtime DLL never contains these diagnostic test exports.
add_library(RbaPresentationTestRuntime SHARED PresentationProbe.cpp PresentationProbe.h CompositionPresenter.cpp CompositionPresenter.h)
target_compile_definitions(RbaPresentationTestRuntime PRIVATE RBA_PRESENT_EXPORTS RBA_COMPOSITION_TESTING)
target_link_libraries(RbaPresentationTestRuntime PRIVATE d3d11 d3d12 dxgi dcomp)
add_executable(PresentationProbeTests PresentationProbeTests.cpp)
add_executable(SourceMaintenanceTests SourceMaintenanceTests.cpp)
target_link_libraries(SourceMaintenanceTests PRIVATE RbaPresentationTestRuntime d3d11 dxgi user32)
add_executable(CompositionGpuTests CompositionGpuTests.cpp CompositionPresenter.cpp)
target_compile_definitions(CompositionGpuTests PRIVATE RBA_COMPOSITION_TESTING)
target_link_libraries(CompositionGpuTests PRIVATE d3d11 d3d12 dxgi dcomp)
target_link_libraries(PresentationProbeTests PRIVATE GfxPluginReduxBetterAAPresentation)
add_library(RbaChildWindowHost STATIC ChildWindowHost.cpp ChildWindowHost.h)
target_include_directories(RbaChildWindowHost PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(RbaChildWindowHost PUBLIC user32)
add_executable(ChildWindowHostTests ChildWindowHostTests.cpp)
target_link_libraries(ChildWindowHostTests PRIVATE RbaChildWindowHost)
foreach(target GfxPluginReduxBetterAAPresentation RbaPresentationTestRuntime PresentationProbeTests SourceMaintenanceTests CompositionGpuTests RbaChildWindowHost ChildWindowHostTests)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
    target_include_directories(${target} SYSTEM PRIVATE "${UNITY_PLUGIN_API}")
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()
target_compile_definitions(GfxPluginReduxBetterAAPresentation PRIVATE RBA_PRESENT_EXPORTS)
enable_testing()
add_test(NAME PresentationProbeTests COMMAND PresentationProbeTests)
add_test(NAME ChildWindowHostTests COMMAND ChildWindowHostTests)
set_tests_properties(ChildWindowHostTests PROPERTIES TIMEOUT 30)
option(RBA_BUILD_FRAME_GENERATION_COORDINATOR "Build the optional owned Unity FG coordinator (not packaged or enabled automatically)" OFF)
if(RBA_BUILD_FRAME_GENERATION_COORDINATOR)
    add_library(GfxPluginReduxBetterAAFrameGeneration SHARED FrameGenerationCoordinator.cpp FrameGenerationCoordinator.h
        ../FrameGeneration/FrameGenerationInputPool.cpp)
    target_link_libraries(GfxPluginReduxBetterAAFrameGeneration PRIVATE RbaChildWindowHost d3d11 d3d12 dxgi)
    target_compile_definitions(GfxPluginReduxBetterAAFrameGeneration PRIVATE RBA_FG_COORDINATOR_EXPORTS RBA_FG_INPUT_STATIC)
    add_executable(FrameGenerationCoordinatorTests FrameGenerationCoordinatorTests.cpp)
    target_link_libraries(FrameGenerationCoordinatorTests PRIVATE GfxPluginReduxBetterAAFrameGeneration)
    add_executable(FrameGenerationCoordinatorLifetimeTests FrameGenerationCoordinatorLifetimeTests.cpp)
    target_compile_definitions(FrameGenerationCoordinatorLifetimeTests PRIVATE RBA_FG_COORDINATOR_EXPORTS RBA_FG_INPUT_STATIC)
    target_link_libraries(FrameGenerationCoordinatorLifetimeTests PRIVATE RbaChildWindowHost d3d11 d3d12 dxgi)
    foreach(target GfxPluginReduxBetterAAFrameGeneration FrameGenerationCoordinatorTests FrameGenerationCoordinatorLifetimeTests)
        target_compile_features(${target} PRIVATE cxx_std_17)
        target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
        target_include_directories(${target} SYSTEM PRIVATE "${UNITY_PLUGIN_API}")
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endforeach()
    add_test(NAME FrameGenerationCoordinatorTests COMMAND FrameGenerationCoordinatorTests)
    add_test(NAME FrameGenerationCoordinatorLifetimeTests COMMAND FrameGenerationCoordinatorLifetimeTests)

endif()
option(RBA_PRESENT_GPU_TESTS "Register hardware GPU tests (requires a coordinated GPU window)" OFF)
if(RBA_PRESENT_GPU_TESTS)
    add_test(NAME SourceMaintenanceTests COMMAND SourceMaintenanceTests)
    add_test(NAME CompositionGpuTests COMMAND CompositionGpuTests)
    set_tests_properties(SourceMaintenanceTests CompositionGpuTests PROPERTIES TIMEOUT 60)
