cmake_minimum_required(VERSION 3.24)
project(ReduxBetterAA_AmdFrameGeneration LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The frame generation prototype requires Windows x64")
endif()
set(FFX_SDK_ROOT "" CACHE PATH "External pinned FidelityFX-SDK checkout")
file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../.." candidate_root)
if(EXISTS "${candidate_root}/AGENTS.md")
    set(source_root "${candidate_root}")
else()
    file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}" source_root)
endif()
file(REAL_PATH "${FFX_SDK_ROOT}" sdk_root)
file(REAL_PATH "${CMAKE_BINARY_DIR}" build_root)
cmake_path(IS_PREFIX source_root "${sdk_root}" NORMALIZE sdk_inside_source)
cmake_path(IS_PREFIX source_root "${build_root}" NORMALIZE build_inside_source)
if(sdk_inside_source OR build_inside_source)
    message(FATAL_ERROR "SDK checkout and build output must remain outside the Better AA source checkout")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${FFX_SDK_ROOT}" rev-parse HEAD
    OUTPUT_VARIABLE sdk_commit OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT sdk_commit STREQUAL "60f4ea81909200d8542eca14dccb2628b763a9a3")
    message(FATAL_ERROR "Requires FidelityFX-SDK 2.3.0 commit 60f4ea81909200d8542eca14dccb2628b763a9a3")
endif()
if(NOT EXISTS "${FFX_SDK_ROOT}/Kits/FidelityFX/framegeneration/include/ffx_framegeneration.h")
    message(FATAL_ERROR "Materialize Kits/FidelityFX/framegeneration/include in the external SDK checkout")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${FFX_SDK_ROOT}" diff --exit-code HEAD --
    Kits/FidelityFX/api/include Kits/FidelityFX/framegeneration/include
    COMMAND_ERROR_IS_FATAL ANY OUTPUT_QUIET)
add_library(ReduxBetterAA.AmdFrameGeneration SHARED AmdFrameGeneration.cpp AmdFrameGeneration.h
    AmdFrameGenerationD3D11.cpp AmdFrameGenerationD3D11.h
    AmdFrameGenerationProvider.cpp AmdFrameGenerationProvider.h AmdFrameGenerationProviderCamera.h)
target_compile_definitions(ReduxBetterAA.AmdFrameGeneration PRIVATE RBA_AMD_FG_EXPORTS UNICODE _UNICODE)
target_include_directories(ReduxBetterAA.AmdFrameGeneration SYSTEM PRIVATE
    "${FFX_SDK_ROOT}/Kits/FidelityFX/api/include" "${FFX_SDK_ROOT}/Kits/FidelityFX/framegeneration/include")
target_link_libraries(ReduxBetterAA.AmdFrameGeneration PRIVATE d3d11 d3d12 dxgi bcrypt user32)
add_executable(AmdFrameGenerationTests AmdFrameGenerationTests.cpp)
target_link_libraries(AmdFrameGenerationTests PRIVATE ReduxBetterAA.AmdFrameGeneration d3d12 dxgi)
add_executable(AmdFrameGenerationD3D11Tests AmdFrameGenerationD3D11Tests.cpp)
target_link_libraries(AmdFrameGenerationD3D11Tests PRIVATE ReduxBetterAA.AmdFrameGeneration d3d11 d3d12 dxgi)
add_executable(AmdFrameGenerationProviderTests AmdFrameGenerationProviderTests.cpp)
target_link_libraries(AmdFrameGenerationProviderTests PRIVATE ReduxBetterAA.AmdFrameGeneration d3d11 d3d12 dxgi user32)
add_library(ReduxBetterAA.FrameGenerationInputs STATIC FrameGenerationInputPool.cpp FrameGenerationInputPool.h)
target_compile_definitions(ReduxBetterAA.FrameGenerationInputs PUBLIC RBA_FG_INPUT_STATIC)
target_link_libraries(ReduxBetterAA.FrameGenerationInputs PUBLIC d3d11 d3d12 dxgi)
add_executable(FrameGenerationInputPoolTests FrameGenerationInputPoolTests.cpp)
target_link_libraries(FrameGenerationInputPoolTests PRIVATE ReduxBetterAA.FrameGenerationInputs)
foreach(target ReduxBetterAA.AmdFrameGeneration AmdFrameGenerationTests AmdFrameGenerationD3D11Tests
    ReduxBetterAA.FrameGenerationInputs FrameGenerationInputPoolTests AmdFrameGenerationProviderTests)
    target_compile_features(${target} PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endforeach()
enable_testing()
add_test(NAME AmdFrameGenerationGpu COMMAND AmdFrameGenerationTests
    "${FFX_SDK_ROOT}/Kits/FidelityFX/signedbin")
add_test(NAME AmdFrameGenerationD3D11Gpu COMMAND AmdFrameGenerationD3D11Tests
    "${FFX_SDK_ROOT}/Kits/FidelityFX/signedbin")
add_test(NAME FrameGenerationInputPoolGpu COMMAND FrameGenerationInputPoolTests)
add_test(NAME AmdFrameGenerationProviderGpu COMMAND AmdFrameGenerationProviderTests
    "${FFX_SDK_ROOT}/Kits/FidelityFX/signedbin")
